        function calculateAndDrawRoutes(start, end) {

            endMarker = L.marker([end.lat, end.lon], { icon: endIcon }).bindPopup("<b>Destination:</b> " + end.name).addTo(map);

            const dLat = (end.lat - start.lat) * 111000;
            const dLon = (end.lon - start.lon) * 85000;
            const directDist = Math.sqrt(dLat*dLat + dLon*dLon);

            // Fast Shortcut Route
            const fastMidLat = (start.lat + end.lat) / 2 + (Math.random() * 0.0003 - 0.00015);
            const fastMidLon = (start.lon + end.lon) / 2 + (Math.random() * 0.0003 - 0.00015);
            const fastCoords = [
                [start.lat, start.lon],
                [start.lat + (end.lat - start.lat)*0.3, start.lon + (end.lon - start.lon)*0.25],
                [fastMidLat, fastMidLon],
                [start.lat + (end.lat - start.lat)*0.75, start.lon + (end.lon - start.lon)*0.8],
                [end.lat, end.lon]
            ];
            const fastMeters = Math.round(directDist * 1.12);
            const fastMins = Math.max(2, Math.round(fastMeters / 75));

            // Safe Well-Lit Route
            const midAxisLon = (start.lon + end.lon) / 2 + 0.0020;
            const safeCoords = [
                [start.lat, start.lon],
                [start.lat, midAxisLon],
                [start.lat + (end.lat - start.lat)*0.35, midAxisLon],
                [end.lat + (start.lat - end.lat)*0.25, midAxisLon],
                [end.lat, midAxisLon],
                [end.lat, end.lon]
            ];
            const safeMeters = Math.round(directDist * 1.32);
            const safeMins = Math.max(fastMins + 2, Math.round(safeMeters / 75));

            safePolyline = L.polyline(safeCoords, { color: '#10b981', weight: 6, opacity: 0.95 }).addTo(map);
            fastPolyline = L.polyline(fastCoords, { color: '#f59e0b', weight: 4, opacity: 0.85, dashArray: '8, 8' }).addTo(map);

            document.getElementById('routeHeaderTitle').innerText = "📍 " + start.name + " ➔ " + end.name;
            
            document.getElementById('safeTime').innerText = safeMins + " min";
            document.getElementById('safeDist').innerText = (safeMeters > 1000 ? (safeMeters/1000).toFixed(1) + " km" : safeMeters + " m");
            document.getElementById('safeScore').innerText = "9.2 / 10";
            document.getElementById('safeDesc').innerText = "Via illuminated main arteries: High foot traffic, active storefronts, continuous street lighting.";

            document.getElementById('fastTime').innerText = fastMins + " min";
            document.getElementById('fastDist').innerText = (fastMeters > 1000 ? (fastMeters/1000).toFixed(1) + " km" : fastMeters + " m");
            document.getElementById('fastScore').innerText = "4.4 / 10";
            document.getElementById('fastDesc').innerText = "Via narrow back-alleys: " + (safeMins - fastMins) + " min faster, but dimly lit and deserted late at night.";

            updateGoogleMapsButton(selectedRouteMode === "safe" ? safeCoords : fastCoords);

            const group = L.featureGroup([safePolyline, fastPolyline]);
            map.fitBounds(group.getBounds(), { padding: [40, 40] });
        }
